// 症状向导页面 - 超简化测试版本
var app = getApp()

var SYMPTOM_CATEGORIES = [
  {
    name: "消化系统",
    symptoms: [
      { label: "呕吐", key: "呕吐" },
      { label: "腹泻", key: "腹泻" },
      { label: "便秘", key: "便秘" },
      { label: "食欲不振", key: "食欲不振" }
    ]
  },
  {
    name: "呼吸系统",
    symptoms: [
      { label: "咳嗽", key: "咳嗽" },
      { label: "打喷嚏", key: "打喷嚏" },
      { label: "呼吸困难", key: "呼吸困难" }
    ]
  }
]

Page({
  data: {
    currentStep: 1,
    petList: [],
    selectedPet: '',
    symptomCategories: SYMPTOM_CATEGORIES,
    selectedSymptoms: [],  // 确保初始化为空数组
    description: ''
  },

  onLoad: function() {
    console.log('=== 症状向导页面加载 ===')
    this.loadMockPetList()
  },

  loadMockPetList: function() {
    var mockPets = [
      { petId: 'mock_001', name: '测试宠物1', type: 'cat', age: 12, breed: '英短' },
      { petId: 'mock_002', name: '测试宠物2', type: 'dog', age: 24, breed: '金毛' }
    ]

    this.setData({
      petList: mockPets
    })
  },

  selectPet: function(e) {
    var petId = e.currentTarget.dataset.petId
    this.setData({
      selectedPet: petId
    })
  },

  addPet: function() {
    wx.navigateTo({
      url: '/pages/pet/profile?action=add'
    })
  },

  // 超简化版本的toggleSymptom - 用于测试
  toggleSymptom: function(e) {
    console.log('=== toggleSymptom 被调用 ===')
    console.log('event:', e)

    var symptomKey = e.currentTarget.dataset.symptomKey
    console.log('symptomKey:', symptomKey)

    var currentSelected = this.data.selectedSymptoms || []
    console.log('当前选中症状:', currentSelected)

    // 最简单的方式：直接操作
    var newSelected = []
    var found = false
    for (var i = 0; i < currentSelected.length; i++) {
      if (currentSelected[i] === symptomKey) {
        found = true
      } else {
        newSelected.push(currentSelected[i])
      }
    }

    if (!found) {
      newSelected.push(symptomKey)
    }

    console.log('更新后症状:', newSelected)

    this.setData({
      selectedSymptoms: newSelected
    })

    console.log('setData 完成')
  },

  onDescriptionInput: function(e) {
    this.setData({
      description: e.detail.value
    })
  },

  nextStep: function() {
    if (this.data.currentStep === 1) {
      if (!this.data.selectedPet) {
        wx.showToast({
          title: '请选择宠物',
          icon: 'none'
        })
        return
      }
    } else if (this.data.currentStep === 2) {
      if (this.data.selectedSymptoms.length === 0) {
        wx.showToast({
          title: '请至少选择一个症状',
          icon: 'none'
        })
        return
      }
    } else if (this.data.currentStep === 3) {
      this.submitAssessment()
      return
    }

    this.setData({
      currentStep: this.data.currentStep + 1
    })
  },

  prevStep: function() {
    if (this.data.currentStep > 1) {
      this.setData({
        currentStep: this.data.currentStep - 1
      })
    }
  },

  submitAssessment: function() {
    if (this.data.selectedSymptoms.length === 0) {
      wx.showToast({
        title: '请至少选择一个症状',
        icon: 'none'
      })
      return
    }

    try {
      wx.showLoading({ title: '评估中...' })

      setTimeout(function() {
        wx.hideLoading()

        var riskLevel = 'low'
        var highRiskSymptoms = ['抽搐', '呼吸困难', '尿血', '排尿困难']
        var midRiskSymptoms = ['呕吐', '腹泻', '咳嗽', '食欲不振', '精神萎靡']

        for (var i = 0; i < this.data.selectedSymptoms.length; i++) {
          var symptom = this.data.selectedSymptoms[i];
          for (var j = 0; j < highRiskSymptoms.length; j++) {
            if (symptom === highRiskSymptoms[j]) {
              riskLevel = 'high'
              break
            }
          }
          if (riskLevel === 'low') {
            for (var k = 0; k < midRiskSymptoms.length; k++) {
              if (symptom === midRiskSymptoms[k]) {
                riskLevel = 'mid'
                break
              }
            }
          }
          if (riskLevel === 'high') {
            break
          }
        }

        var foundPet = null;
        for (var i = 0; i < this.data.petList.length; i++) {
          if (this.data.petList[i].petId === this.data.selectedPet) {
            foundPet = this.data.petList[i];
            break;
          }
        }
        var petName = foundPet ? foundPet.name : '宠物';

        wx.navigateTo({
          url: '/pages/risk/result?riskLevel=' + riskLevel + '&petName=' + petName
        })

      }.bind(this), 1000)

    } catch (error) {
      wx.hideLoading()
      console.error('提交评估失败:', error)
      wx.showToast({
        title: '提交失败，请重试',
        icon: 'none'
      })
    }
  }
})
