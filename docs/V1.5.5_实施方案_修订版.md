## 4. 修订后的云函数设计

### 4.1 syncDailyContent（独立机器鉴权）

**核心改动**：不再 `verifyToken`，改用独立的 `verifyMachineToken`。Python 侧 token 格式 `{ts}.{hmac}` 不变，只换密钥。

```javascript
const cloud = require('wx-server-sdk');
const crypto = require('crypto');
const { COLLECTIONS, RESPONSE_CODE, SERVER_CONFIG, warmupConfig } = require('./common/constants');
const { createLogger } = require('./common/logger');

cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV });
const db = cloud.database();
const logger = createLogger('syncDailyContent');

// === 独立机器鉴权（P0-1 修正）===
// 密钥来自 system_config.content_push_secret，与终端用户 TOKEN_SECRET 物理隔离
function getMachineSecret() {
  const s = SERVER_CONFIG.CONTENT_PUSH_SECRET;
  if (!s) throw new Error('[syncDailyContent] CONTENT_PUSH_SECRET 未配置');
  return s;
}

// token 格式：{秒级时间戳}.{hmac_sha256(secret, 时间戳)}，与 push_daily_content.py 对齐
function verifyMachineToken(token) {
  if (!token || typeof token !== 'string') return false;
  const parts = token.split('.');
  if (parts.length !== 2) return false;
  const ts = Number(parts[0]);
  if (!Number.isFinite(ts)) return false;
  // 5 分钟窗口防重放（P1-6）
  const now = Math.floor(Date.now() / 1000);
  if (Math.abs(now - ts) > 300) return false;
  const expected = crypto.createHmac('sha256', getMachineSecret()).update(parts[0]).digest('hex');
  const a = Buffer.from(parts[1], 'hex');
  const b = Buffer.from(expected, 'hex');
  return a.length === b.length && crypto.timingSafeEqual(a, b);
}

exports.main = async (event, context) => {
  await warmupConfig(db);
  // token 兼容 event.token 与 header 两种传法
  const token = event.token || (event.headers && event.headers['x-machine-token']);
  if (!verifyMachineToken(token)) {
    logger.warn('机器鉴权失败');
    return { code: RESPONSE_CODE.UNAUTHORIZED, msg: '身份验证失败', data: {} };
  }

  const { title, summary, content, cover_url, category, target_pet } = event;
  if (!title || !content) {
    return { code: RESPONSE_CODE.ERROR, msg: '缺少标题或内容', data: {} };
  }

  const today = cnDateStr(); // UTC+8，见 §4.8

  try {
    const existing = await db.collection(COLLECTIONS.KNOWLEDGE_ARTICLES)
      .where({ title }).limit(1).get();

    if (existing.data.length > 0) {
      const id = existing.data[0]._id;
      await db.collection(COLLECTIONS.KNOWLEDGE_ARTICLES).doc(id).update({
        data: {
          featured_date: today,
          summary: summary || existing.data[0].summary,
          content: content || existing.data[0].content,
          cover_image: cover_url || existing.data[0].cover_image,
          updated_at: new Date(),
        },
      });
      return { code: 0, msg: '已更新为今日推荐', data: { article_id: id } };
    }

    const result = await db.collection(COLLECTIONS.KNOWLEDGE_ARTICLES).add({
      data: {
        title,
        summary: summary || '',
        content,
        cover_image: cover_url || DEFAULT_COVER,
        category: category || 'digestive',
        target_pet: target_pet || 'all',
        petType: target_pet || 'all',
        status: 'published',
        featured_date: today,        // ← 靠 featured_date 选中，不再依赖 sort_order
        sort_order: 999,
        view_count: 0,
        viewCount: 0,
        member_only: false,
        publishedAt: new Date(),
      },
    });
    return { code: 0, msg: '新增成功', data: { article_id: result._id } };
  } catch (err) {
    logger.error('syncDailyContent 失败', err.message);
    return { code: RESPONSE_CODE.SERVER_ERROR, msg: err.message, data: {} };
  }
};

const DEFAULT_COVER = 'https://well-tools-d8gwhxdlcf84937ed-1435021389.tcloudbaseapp.com/knowledge-covers/default.jpg';
```

### 4.2 generatePetDiary（分片 + 并发 + 活跃过滤 + 灰度 + 幂等）

**五项关键修正**：①单实例分批 + 批内并发 + 配足超时（P0-5）②仅活跃用户（P1-1）③确定性灰度分组（P1-5）④唯一索引幂等（P1-3）⑤DeepSeek 纯文本润色（P0-2）。

```javascript
const cloud = require('wx-server-sdk');
const { COLLECTIONS, SERVER_CONFIG, warmupConfig } = require('./common/constants');
const { createLogger } = require('./common/logger');
const { callDeepSeekAPI } = require('./common/report-engine'); // 见 §4.7.1 重构后
const { filterSensitive } = require('./common/sensitiveWords'); // 见 §4.7.2
const templates = require('./diary-templates');

cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV });
const db = cloud.database();
const _ = db.command;
const logger = createLogger('generatePetDiary');

const ACTIVE_DAYS = 14;        // P1-1 只生成近 14 天活跃用户
const CONCURRENCY = 10;        // 批内并发
const DIARY_AI_TIMEOUT_MS = 8000;

// === 确定性灰度分组（P1-5）===
function hashStr(s) {
  let h = 0;
  for (let i = 0; i < s.length; i++) h = ((h << 5) - h + s.charCodeAt(i)) | 0;
  return Math.abs(h);
}
function inTreatment(userId, rolloutPercent) {
  return hashStr(userId) % 100 < rolloutPercent;
}

async function generateForPet(pet, cohort, aiEnabled) {
  const userId = pet.user_id;
  const today = cnDateStr();

  // 幂等：先查（唯一索引是最终兜底）
  const exists = await db.collection(COLLECTIONS.PET_MOMENTS)
    .where({ user_id: userId, date: today }).limit(1).get();
  if (exists.data.length > 0) return { skipped: true };

  // 昨日行为（UTC+8 边界）
  const yr = cnYesterdayRange();
  const symptomRes = await db.collection(COLLECTIONS.SYMPTOM_RECORDS)
    .where({ user_id: userId, created_at: _.gte(yr.start).lt(yr.end) }).limit(1).get();
  const knowledgeCount = await db.collection(COLLECTIONS.ANALYTICS_EVENTS)
    .where({ user_id: userId, event_name: 'knowledge_view', created_at: _.gte(yr.start).lt(yr.end) }).count();

  const isFirst = (await db.collection(COLLECTIONS.PET_MOMENTS)
    .where({ user_id: userId }).count()).total === 0;

  const templateKey = selectTemplateKey(isFirst, symptomRes.data.length, knowledgeCount.total);
  const tags = pet.personality_tags || [];
  const templateText = templates.pick(templateKey, pet.type, tags);

  const vars = {
    pet_name: pet.name || (pet.type === 'cat' ? '猫咪' : '狗狗'),
    pet_type: pet.type === 'cat' ? '猫咪' : '狗狗',
    symptoms: symptomRes.data[0] && symptomRes.data[0].symptom_names
      ? symptomRes.data[0].symptom_names.join('、') : '',
    days: daysSince(pet.created_at),
  };
  let content = fillTemplate(templateText, vars);
  let aiUsed = false;

  // AI 润色（冷启动 welcome 不调 AI 控成本）
  if (aiEnabled && templateKey !== 'welcome') {
    try {
      const polished = await withTimeout(
        callDeepSeekAPI(
          '你是一只' + vars.pet_type + '，用第一人称写一句100字以内的日记。语气温暖可爱自然，直接输出日记正文。',
          content,
          { temperature: 0.8, maxTokens: 200, responseFormat: null } // P0-2：纯文本
        ),
        DIARY_AI_TIMEOUT_MS
      );
      if (polished && polished.length >= 10 && polished.length <= 200) {
        content = polished;
        aiUsed = true;
      }
    } catch (e) {
      logger.warn('AI 润色失败，用模板：' + e.message);
    }
  }

  // 敏感词过滤（care 语境豁免医学术语，P1-7）
  content = filterSensitive(content, templateKey);

  // 写入（唯一索引兜底重复）
  try {
    await db.collection(COLLECTIONS.PET_MOMENTS).add({
      data: {
        user_id: userId, pet_id: pet._id, date: today, content,
        type: 'diary', source: 'auto',
        mood: pickMood(templateKey), template_used: templateKey,
        is_first: isFirst, diary_cohort: cohort, ai_used: aiUsed,
        created_at: new Date(),
      },
    });
  } catch (e) {
    if (e.errCode === -502001 || /duplicate/i.test(e.message)) return { skipped: true };
    throw e;
  }

  // 订阅消息（配额机制见 §6）
  await trySendDiarySubscribe(userId, pet, content);
  return { ok: true };
}

exports.main = async (event, context) => {
  await warmupConfig(db);
  const rollout = Number(SERVER_CONFIG.DIARY_ROLLOUT_PERCENT || 0);
  const aiEnabled = rollout > 0; // rollout=0 时不生成、不调 AI

  // P1-1 活跃过滤：只取近 14 天活跃用户（避免遍历历史注册全量）
  const cutoff = new Date(Date.now() - ACTIVE_DAYS * 86400000);
  const activeUsers = await db.collection(COLLECTIONS.USERS)
    .where({ last_active_at: _.gte(cutoff) })
    .field({ user_id: true }).limit(2000).get();
  const userIds = activeUsers.data.map((u) => u.user_id).filter(Boolean);
  logger.info('活跃用户数：' + userIds.length + '，rollout=' + rollout);

  if (userIds.length === 0) return { code: 0, data: { ok: 0, skipped: 0, errors: 0 } };

  // 拉这些用户的 pets（分页查，每页 1000）
  let allPets = [];
  for (let i = 0; i < userIds.length; i += 1000) {
    const chunk = userIds.slice(i, i + 1000);
    const r = await db.collection(COLLECTIONS.PETS)
      .where({ user_id: _.in(chunk) }).limit(1000).get();
    allPets = allPets.concat(r.data);
  }

  let ok = 0, skipped = 0, errors = 0;
  // P0-5：批内并发(CONCURRENCY=10)、批间串行——单实例配足超时即可跑完验证期规模
  for (let i = 0; i < allPets.length; i += CONCURRENCY) {
    const batch = allPets.slice(i, i + CONCURRENCY);
    const results = await Promise.all(batch.map((pet) => {
      // 确定性灰度：control 组不生成（保留作 H1 基线）
      if (!inTreatment(pet.user_id, rollout)) return Promise.resolve({ skipped: true });
      return generateForPet(pet, 'treatment', aiEnabled)
        .catch((e) => { logger.error('生成失败 ' + pet.user_id, e.message); return { err: true }; });
    }));
    results.forEach((r) => {
      if (r.ok) ok++;
      else if (r.skipped) skipped++;
      else errors++;
    });
  }
  logger.info('完成 ok=' + ok + ' skipped=' + skipped + ' err=' + errors);
  return { code: 0, data: { ok, skipped, errors, active: userIds.length } };
};
```

> **control 组的处理**：H1 要对比"有日记组 vs 无日记组"。`inTreatment` 是**确定性哈希**（同一 openid 永远落同一组），control 组不生成日记。留存分析时用**同一 `hashStr` 公式**在分析端/脚本里重算分组即可，不必依赖 `users.diary_cohort` 预落库（§3.4 的该字段为可选缓存，仅方便管理后台直查，由 generatePetDiary 顺手 upsert，缺失也不影响分析）。

**定时触发器**（北京时间 00:30）：
```jsonc
{ "triggers": [{ "name": "dailyDiary", "type": "timer", "config": "0 30 0 * * * *" }] }
```
> 微信云函数定时器按 **UTC** 解释 cron。北京 00:30 = UTC 16:30，故 cron 应配 `0 30 16 * * * *`（或在云函数环境设 `TZ=Asia/Shanghai` 后用 `0 30 0`）。以实际云环境验证为准——**这是 P1-2 时区坑的具体表现，部署时务必确认**。

**超时配置（P0-5 必做）**：云开发云函数默认超时较短，需在 `config.json` 把 `generatePetDiary` 的 `timeout` 调到**上限**（微信云开发函数最大 60s）。验证期规模（活跃 500 × rollout 20% ≈ 100 个 treatment 用户，批内并发 10、DeepSeek 8s 兜底）单实例 60s 足够。

**规模化演进（验证通过后再考虑）**：当 treatment 用户量增长到单实例 60s 跑不完时，再引入"队列集合 + 高频消费者定时器"或 `cloud.callFunction` 分片（**注意** `cloud.callFunction` 返回 Promise、是同步 RPC 语义，调用方返回后被调实例不保证继续执行，不能简单 fire-and-forget；需 await 各分片或用队列解耦）。验证期不引入此复杂度。

**辅助函数**（`withTimeout` / `selectTemplateKey` / `fillTemplate` / `pickMood` / `daysSince` / `trySendDiarySubscribe`）：`withTimeout` 用 `Promise.race` 包一个 8s reject；其余逻辑同原方案，`trySendDiarySubscribe` 见 §6。

// 统一 UTC+8，消除云函数默认 UTC 的日期错位
function cnNow() { return new Date(Date.now() + 8 * 3600 * 1000); }
function cnDateStr() {
  const d = cnNow();
  return d.getUTCFullYear() + '-' +
    String(d.getUTCMonth() + 1).padStart(2, '0') + '-' +
    String(d.getUTCDate()).padStart(2, '0');
}
function cnYesterdayRange() {
  const startUtc = Date.now() - 86400000;
  const start = new Date(startUtc);
  start.setUTCHours(16, 0, 0, 0); // 北京 00:00 = UTC 16:00
  const end = new Date(start.getTime() + 86400000);
  return { start, end };
}
module.exports = { cnNow, cnDateStr, cnYesterdayRange };

微信订阅消息默认"一次授权 = 一次发送配额"，且**服务端无法查询剩余配额**（只能 send 后据返回码判断）。机制设计：

### 6.1 多触点累加配额（前端）

不再只在创建宠物时授权一次，但**每个触点必须由用户的明确点击行为触发**——微信要求 `requestSubscribeMessage` 在点击交互期间调用，**禁止在 onLoad/onShow 自动弹**（qw-2：自动弹或频繁弹属"诱导授权"，易被用户投诉导致订阅消息能力被限制）。合规触点：
- 创建宠物"保存"按钮点击时（原 T18，与 savePet 并行触发，不要在 savePet 异步回调里弹——会脱离点击上下文）
- 症状自查结果页"开启日记提醒"CTA 按钮
- 日记详情页"明天提醒我"按钮
- 知识详情页底部 CTA

用户每 `accept` 一次 → `diary_subscribe_quota +1`。**文案合规**：必须如实说明"将发送日记提醒"，不得用"立即领取/否则丢失"等诱导话术。

```javascript
function requestDiarySubscribe() {
  wx.requestSubscribeMessage({
    tmplIds: [DIARY_TEMPLATE_ID],
    success(res) {
      if (res[DIARY_TEMPLATE_ID] === 'accept') {
        wx.cloud.callFunction({
          name: 'saveUserProfile',
          data: { action: 'diary_subscribe_accept' }, // 服务端 +1
        });
      }
    },
  });
}
```

### 6.2 服务端按返回码校准

```javascript
async function trySendDiarySubscribe(userId, pet, content) {
  const u = await db.collection(COLLECTIONS.USERS).where({ user_id: userId }).limit(1).get();
  if (!u.data.length) return;
  const quota = u.data[0].diary_subscribe_quota || 0;
  if (quota <= 0) return; // 无配额不尝试（省调用）

  try {
    await cloud.openapi.subscribeMessage.send({
      touser: userId,
      templateId: SERVER_CONFIG.DIARY_TEMPLATE_ID,
      page: 'pages/index/index',
      data: {
        thing1: { value: (pet.name || '毛孩子') + '的今日日记' },
        thing2: { value: content.substring(0, 20) },
      },
    });
    // 发送成功 → 配额 -1
    await db.collection(COLLECTIONS.USERS).doc(u.data[0]._id)
      .update({ data: { diary_subscribe_quota: _.inc(-1) } });
  } catch (e) {
    if (e.errCode === 43101) {
      // 用户未订阅/配额耗尽 → 配额归零校准
      await db.collection(COLLECTIONS.USERS).doc(u.data[0]._id)
        .update({ data: { diary_subscribe_quota: 0 } });
    }
    // 其他错误静默
  }
}
```

### 6.3 H3 口径修订（重要）

原 H3"召回"假设无法在机制上持续多日。**修订口径**：

| 子假设 | 原口径 | 修订口径 |
|--------|--------|---------|
| H3-a | 授权率 ≥40%（持续） | **累计触点授权转化率**：在 N 个触点中至少 accept 1 次的用户占比 ≥40% |
| H3-b | 打开率 ≥15%（持续召回） | **单次触达→当次打开率**：发出后 24h 内打开 ≥15%（不再宣称"持续召回"） |

> 持续召回留待"长期订阅类目"申请通过后再验证；当前版本只验证"单次触达有效性"。

### 7.2 留存口径（Week4 看板必须实现）

以 `analytics_events` 构造"用户×天活跃事实表"：

```sql
-- 次日留存（按 diary_cohort 切分）
WITH active AS (
  SELECT user_id, DATE(created_at) AS d
  FROM analytics_events WHERE event_name = 'app_session' GROUP BY user_id, DATE(created_at)
)
SELECT u.diary_cohort,
  COUNT(DISTINCT a1.user_id) AS d0,
  COUNT(DISTINCT a2.user_id) AS d1,
  ROUND(COUNT(DISTINCT a2.user_id) * 100.0 / NULLIF(COUNT(DISTINCT a1.user_id),0), 1) AS d1_retention
FROM active a1
LEFT JOIN active a2 ON a2.user_id = a1.user_id AND a2.d = a1.d + INTERVAL 1 DAY
JOIN users u ON u.user_id = a1.user_id
WHERE a1.d = :launch_date
GROUP BY u.diary_cohort;
```

> 对比 `treatment` vs `control` 两组的 d1_retention，即 H1 的判定依据。

> **实验周期封顶（ds-3）**：control 组长期被剥夺日记，若日记确有价值，其留存会低于自然态，反而**放大** treatment-control 差值（夸大 H1 效果）。故验证周期严格 ≤4 周；H1 一旦通过，**立即给 control 组补开日记**，不长期维持对照组。

## 10. 成本重估（P1-1）

| 项 | 原估算 | 修订估算（基于活跃量） |
|----|--------|----------------------|
| DeepSeek | DAU 500 × 1 次 | **活跃用户数 × treatment 占比 × AI 比例**。若活跃 500、rollout 20%、AI 30% → 500×20%×30% = 30 次/天 ≈ 900 次/月 |
| 范围 | 全表 | 仅近 14 天活跃 + control 组不调 AI |
| 月增量 | <50 元 | <10 元（AI 真实调用量大幅下降；但若活跃用户增长到 5000，则 ~100 元——**随活跃量线性**） |

> 关键：成本不再因"遍历历史全量"而失真，且随灰度比例可线性控制。
