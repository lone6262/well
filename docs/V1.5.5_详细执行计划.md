### T2: 创建 syncDailyContent 云函数

**新建目录**：`cloudfunctions/syncDailyContent/`

**新建文件**：

**index.js 核心逻辑**：

```javascript
const cloud = require('wx-server-sdk');
const { COLLECTIONS, RESPONSE_CODE, warmupConfig } = require('./common/constants');
const { verifyToken } = require('./common/auth');

cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV });
const db = cloud.database();

exports.main = async (event, context) => {
  await warmupConfig(db);

  // 鉴权：复用 HMAC Token（内容工厂侧配同密钥）
  if (!verifyToken(event.token)) {
    return { code: RESPONSE_CODE.UNAUTHORIZED, msg: '身份验证失败', data: {} };
  }

  const { title, summary, content, cover_url, category, target_pet } = event;

  if (!title || !content) {
    return { code: RESPONSE_CODE.ERROR, msg: '缺少标题或内容', data: {} };
  }

  const today = new Date().toISOString().slice(0, 10); // YYYY-MM-DD

  try {
    // 查是否已有同标题文章
    const existing = await db.collection(COLLECTIONS.KNOWLEDGE_ARTICLES)
      .where({ title }).limit(1).get();

    if (existing.data.length > 0) {
      // 更新 featured_date
      const articleId = existing.data[0]._id;
      await db.collection(COLLECTIONS.KNOWLEDGE_ARTICLES).doc(articleId).update({
        data: {
          featured_date: today,
          summary: summary || existing.data[0].summary,
          content: content || existing.data[0].content,
          cover_image: cover_url || existing.data[0].cover_image,
          updated_at: new Date()
        }
      });
      return { code: 0, msg: '已更新为今日推荐', data: { article_id: articleId } };
    }

    // 新增
    const result = await db.collection(COLLECTIONS.KNOWLEDGE_ARTICLES).add({
      data: {
        title,
        summary: summary || '',
        content,
        cover_image: cover_url || DEFAULT_COVER,
        category: category || 'digestive',
        target_pet: target_pet || 'all',
        petType: target_pet || 'all',
        status: 'published',
        featured_date: today,
        view_count: 0,
        viewCount: 0,
        sort_order: 999,  // 排到后面，靠 featured_date 选中
        member_only: false,
        publish_time: new Date().toISOString(),
        publishedAt: new Date().toISOString()
      }
    });

    return { code: 0, msg: '新增成功', data: { article_id: result._id } };
  } catch (err) {
    return { code: RESPONSE_CODE.SERVER_ERROR, msg: err.message, data: {} };
  }
};

const DEFAULT_COVER = 'https://well-tools-d8gwhxdlcf84937ed-1435021389.tcloudbaseapp.com/knowledge-covers/default.jpg';
```

### T3: 创建 push_daily_content.py 脚本

**新建文件**：`scripts/push_daily_content.py`

**逻辑**：
1. 读 `content/` 目录今天的 `xhs_note_*.md` 或 `gzh_note_*.md`
2. 解析标题和正文（复用 `obsidian_reader.parse_frontmatter`）
3. 生成 HMAC Token（复用 well 项目的 `TOKEN_SECRET`）
4. HTTP POST 调 `syncDailyContent` 云函数

**核心结构**：

```python
"""推送当日内容到小程序首页"""
import sys, os, json, hmac, hashlib, time
sys.path.insert(0, os.path.dirname(__file__))
from obsidian_reader import parse_frontmatter

PROJECT_ROOT = r"F:\codex\自媒体工作流"
CONTENT_DIR = os.path.join(PROJECT_ROOT, "content")

# Token 密钥（与 well/cloudfunctions/common/secrets.js 中 TOKEN_SECRET 一致）
TOKEN_SECRET = os.environ.get("MEWORA_TOKEN_SECRET", "")

CLOUD_FUNC_URL = "https://cloud1-d8gdi44zqfec250b5.api.tcloudbasegateway.com/v1/cloudfunctions/syncDailyContent"

def generate_token():
    ts = str(int(time.time()))
    sig = hmac.new(TOKEN_SECRET.encode(), ts.encode(), hashlib.sha256).hexdigest()
    return f"{ts}.{sig}"

def get_today_note():
    today = time.strftime("%Y-%m-%d")  # 或按文件修改时间判断
    # 找今天的 gzh_note 或 xhs_note
    for f in sorted(os.listdir(CONTENT_DIR)):
        if f.startswith("gzh_note_") and f.endswith(".md"):
            # 取最近修改的
            return os.path.join(CONTENT_DIR, f)
    return None

def main():
    note_path = get_today_note()
    if not note_path:
        print("未找到今日文章")
        return 1

    with open(note_path, "r", encoding="utf-8-sig") as f:
        raw = f.read()
    # 解析标题（第一行 # xxx）
    lines = raw.strip().split("\n")
    title = lines[0].replace("#", "").strip() if lines else "今日科普"
    content = "\n".join(lines[1:]).strip()
    # 摘要取前 150 字
    summary = content[:150].replace("\n", " ") + "..."

    token = generate_token()
    payload = {
        "title": title,
        "summary": summary,
        "content": content,
        "cover_url": "",
        "category": "digestive",
        "target_pet": "all",
        "token": token
    }

    # HTTP POST（用 requests）
    import requests
    resp = requests.post(CLOUD_FUNC_URL, json=payload, timeout=30)
    result = resp.json()
    if result.get("code") == 0:
        print(f"推送成功：{title}")
    else:
        print(f"推送失败：{result.get('msg')}")
    return 0

if __name__ == "__main__":
    sys.stdout.reconfigure(encoding="utf-8")
    sys.exit(main())
```

### T4: 改造 loadDailyKnowledge 查 featured_date

**改动文件**：`well/miniprogram/pages/index/data-loader.js`

**操作**：将 `loadDailyKnowledge` 函数改为优先查 `featured_date = 今天`：

```javascript
function loadDailyKnowledge(pageCtx) {
  if (!app.globalData.cloudDevelopmentAvailable) return

  var today = new Date()
  var todayStr = today.getFullYear() + '-' +
    String(today.getMonth() + 1).padStart(2, '0') + '-' +
    String(today.getDate()).padStart(2, '0')

  // 先查今天推荐的
  wx.cloud.callFunction({
    name: 'getKnowledgeList',
    data: { page: 1, pageSize: 1 },
    success: function(res) {
      if (res.result && res.result.code === 0) {
        let articles = res.result.data.articles || res.result.data.list || []
        // 优先匹配 featured_date
        var featured = articles.find(function(a) { return a.featured_date === todayStr })
        if (featured) {
          pageCtx.setData({ dailyKnowledge: featured })
          return
        }
        // 回退：取第一条
        if (articles.length > 0) {
          pageCtx.setData({ dailyKnowledge: articles[0] })
        }
      }
    },
    fail: function() {}
  })
}
```

注意：`getKnowledgeList` 当前按 `sort_order` 排序，`syncDailyContent` 新建文章时 sort_order 设为 999，所以不会被排在最前。需要 `getKnowledgeList` 返回的文章里带上 `featured_date` 字段，以便前端匹配。

**额外改动**：`getKnowledgeList/index.js` 的 `articles.map` 中增加 `featured_date: article.featured_date || ''`

**index.js 核心逻辑**：

```javascript
const cloud = require('wx-server-sdk');
const { COLLECTIONS, warmupConfig } = require('./common/constants');
const { createLogger } = require('./common/logger');

cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV });
const db = cloud.database();
const _ = db.command;
const logger = createLogger('generatePetDiary');

const { callDeepSeekAPI } = require('./common/report-engine');
const templates = require('./diary-templates');

// 敏感词列表（简化版，实际复用 sensitiveWords.js 的词库）
const SENSITIVE_WORDS = ['激素', '抗生素', '癌症', '肿瘤', '安乐死', '毒药'];

function getTodayStr() {
  var d = new Date();
  return d.getFullYear() + '-' +
    String(d.getMonth() + 1).padStart(2, '0') + '-' +
    String(d.getDate()).padStart(2, '0');
}

function getYesterdayRange() {
  var now = new Date();
  var start = new Date(now);
  start.setDate(start.getDate() - 1);
  start.setHours(0, 0, 0, 0);
  var end = new Date(now);
  end.setHours(0, 0, 0, 0);
  return { start, end };
}

function daysSince(dateStr) {
  if (!dateStr) return 1;
  var diff = Date.now() - new Date(dateStr).getTime();
  return Math.max(1, Math.floor(diff / 86400000));
}

function filterSensitive(text) {
  var cleaned = text;
  SENSITIVE_WORDS.forEach(function(w) {
    cleaned = cleaned.split(w).join('**');
  });
  return cleaned;
}

function pickMood(templateKey) {
  var moods = {
    welcome: 'excited',
    care: 'grateful',
    learn: 'curious',
    daily: 'happy'
  };
  return moods[templateKey] || 'happy';
}

function selectTemplateKey(isFirst, symptomCount, knowledgeCount) {
  if (isFirst) return 'welcome';
  if (symptomCount > 0) return 'care';
  if (knowledgeCount > 0) return 'learn';
  return 'daily';
}

function fillTemplate(templateText, vars) {
  return templateText.replace(/\{(\w+)\}/g, function(match, key) {
    return vars[key] !== undefined ? String(vars[key]) : match;
  });
}

async function generateForPet(pet) {
  var userId = pet.user_id;
  var today = getTodayStr();

  // 幂等检查
  var existing = await db.collection(COLLECTIONS.PET_MOMENTS)
    .where({ user_id: userId, date: today }).count();
  if (existing.total > 0) return;

  var yRange = getYesterdayRange();

  // 拉取昨日行为
  var symptomResult = await db.collection(COLLECTIONS.SYMPTOM_RECORDS)
    .where({ user_id: userId, created_at: _.gte(yRange.start).lt(yRange.end) })
    .limit(1).get();

  var knowledgeResult = await db.collection(COLLECTIONS.ANALYTICS_EVENTS)
    .where({ user_id: userId, event_name: 'knowledge_view',
             created_at: _.gte(yRange.start).lt(yRange.end) }).count();

  // 判断是否首次
  var diaryCount = await db.collection(COLLECTIONS.PET_MOMENTS)
    .where({ user_id: userId }).count();
  var isFirst = diaryCount.total === 0;

  var templateKey = selectTemplateKey(
    isFirst,
    symptomResult.data.length,
    knowledgeResult.total
  );

  // 选模板（带性格标签）
  var tags = pet.personality_tags || [];
  var templateText = templates.pick(templateKey, pet.type, tags);

  // 填充
  var vars = {
    pet_name: pet.name || (pet.type === 'cat' ? '猫咪' : '狗狗'),
    pet_type: pet.type === 'cat' ? '猫咪' : '狗狗',
    symptoms: (symptomResult.data[0] && symptomResult.data[0].symptom_names)
              ? symptomResult.data[0].symptom_names.join('、') : '',
    days: daysSince(pet.created_at)
  };

  var content = fillTemplate(templateText, vars);

  // AI 润色（冷启动模板跳过 AI，控成本）
  if (templateKey !== 'welcome') {
    try {
      var systemPrompt = '你是一只' + vars.pet_type + '，用第一人称写一句100字以内的日记。' +
        '语气温暖、可爱、自然。不要说"作为一只猫"之类的话。直接输出日记内容。';
      var polished = await callDeepSeekAPI(systemPrompt, content);
      if (polished && polished.length > 10 && polished.length < 200) {
        content = polished;
      }
    } catch (e) {
      logger.warn('AI 润色失败，使用模板：' + e.message);
    }
  }

  // 敏感词过滤
  content = filterSensitive(content);

  // 写入
  await db.collection(COLLECTIONS.PET_MOMENTS).add({
    data: {
      user_id: userId,
      pet_id: pet._id,
      date: today,
      content: content,
      type: 'diary',
      source: 'auto',
      mood: pickMood(templateKey),
      template_used: templateKey,
      is_first: isFirst,
      created_at: new Date()
    }
  });

  logger.info('日记生成成功：' + userId + ' ' + today);
}

exports.main = async (event, context) => {
  await warmupConfig(db);
  logger.info('=== 定时日记生成开始 ===');

  // 分批获取所有宠物
  var batch = 100;
  var skip = 0;
  var total = 0;
  var errors = 0;

  while (true) {
    var result = await db.collection(COLLECTIONS.PETS)
      .skip(skip).limit(batch).get();

    if (!result.data || result.data.length === 0) break;

    for (var i = 0; i < result.data.length; i++) {
      try {
        await generateForPet(result.data[i]);
        total++;
      } catch (e) {
        logger.error('生成失败 ' + result.data[i].user_id + ': ' + e.message);
        errors++;
      }
    }

    skip += result.data.length;
    if (result.data.length < batch) break;
  }

  logger.info('完成：成功 ' + total + '，失败 ' + errors);
  return { code: 0, msg: '日记生成完成', data: { success: total, errors: errors } };
};
```

### T6: 创建 getPetDiary 云函数

**新建目录**：`cloudfunctions/getPetDiary/`

**核心逻辑**：

```javascript
const cloud = require('wx-server-sdk');
const { COLLECTIONS, RESPONSE_CODE, warmupConfig } = require('./common/constants');

cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV });
const db = cloud.database();

exports.main = async (event, context) => {
  await warmupConfig(db);
  const { OPENID } = cloud.getWXContext();
  const { page = 1, pageSize = 10 } = event;

  if (!OPENID) {
    return { code: RESPONSE_CODE.UNAUTHORIZED, msg: '用户未登录', data: {} };
  }

  // 查会员状态判断是否限制 7 天
  var memberResult = await db.collection(COLLECTIONS.MEMBERS)
    .where({ user_id: OPENID, status: 'active' }).limit(1).get();
  var isMember = memberResult.data && memberResult.data.length > 0;

  var query = db.collection(COLLECTIONS.PET_MOMENTS)
    .where({ user_id: OPENID })
    .orderBy('date', 'desc');

  // 非会员限制 7 天
  if (!isMember) {
    var sevenDaysAgo = new Date();
    sevenDaysAgo.setDate(sevenDaysAgo.getDate() - 7);
    query = query.where({ user_id: OPENID, created_at: db.command.gte(sevenDaysAgo) });
  }

  var skip = (page - 1) * pageSize;
  var countResult = await query.count();
  var result = await query.skip(skip).limit(pageSize).get();

  var diaries = result.data.map(function(d) {
    return {
      _id: d._id,
      date: d.date,
      content: d.content,
      mood: d.mood,
      is_first: d.is_first,
      created_at: d.created_at
    };
  });

  return {
    code: 0,
    msg: '获取成功',
    data: {
      diaries: diaries,
      total: countResult.total,
      page: page,
      pageSize: pageSize,
      hasMore: skip + pageSize < countResult.total,
      isMember: isMember
    }
  };
};
```

```javascript
function loadPetDiary(pageCtx) {
  if (!app.globalData.cloudDevelopmentAvailable) return

  wx.cloud.callFunction({
    name: 'getPetDiary',
    data: { page: 1, pageSize: 1 },
    success: function(res) {
      if (res.result && res.result.code === 0) {
        var diaries = res.result.data.diaries || []
        if (diaries.length > 0) {
          var d = diaries[0]
          // 格式化日期
          var dateObj = new Date(d.created_at)
          var dateText = (dateObj.getMonth() + 1) + '月' + dateObj.getDate() + '日'
          pageCtx.setData({
            todayDiary: {
              _id: d._id,
              content: d.content,
              dateText: dateText,
              petName: '' // 由 loadPetList 填充宠物名
            }
          })
          // 尝试用宠物列表中的名字
          if (pageCtx.data.petsList && pageCtx.data.petsList.length > 0) {
            pageCtx.setData({
              'todayDiary.petName': pageCtx.data.petsList[0].name
            })
          }
        }
      }
    },
    fail: function() {
      // 静默失败
    }
  })
}
```

```javascript
toggleTag: function(e) {
  var tag = e.currentTarget.dataset.tag
  var tags = this.data.selectedTags || []
  var idx = tags.indexOf(tag)
  if (idx !== -1) {
    tags.splice(idx, 1)
  } else {
    if (tags.length >= 2) {
      wx.showToast({ title: '最多选2个', icon: 'none' })
      return
    }
    tags.push(tag)
```
